```python
from sqlalchemy import inspect, text


def ensure_play_count_column(engine) -> None:
    """Add watch_history.play_count for databases created before v3.0.

    SQLite cannot ALTER COLUMN, but ADD COLUMN is safe and cheap. Guarded by
    an inspector check so re-running is a no-op.
    """
    inspector = inspect(engine)
    if "watch_history" not in inspector.get_table_names():
        return  # create_all will build the full v3 schema
    columns = {col["name"] for col in inspector.get_columns("watch_history")}
    if "play_count" in columns:
        return
    with engine.begin() as conn:
        conn.execute(
            text("ALTER TABLE watch_history ADD COLUMN play_count INTEGER DEFAULT 1")
        )
        conn.execute(text("UPDATE watch_history SET play_count = 1"))
    logger.info("Added play_count column to watch_history")
```

```python
from __future__ import annotations

from ..models import Movie


def movie_to_document(movie: Movie) -> str:
    """Concatenate a movie's text fields into one whitespace-joined document.
```

```python
"""One-shot Jellyfin catalog export to normalized JSON (spec Section 4).

Reads credentials from ~/.config/moviepicker/.env:
    JELLYFIN_URL=http://100.112.3.116:8096
    JELLYFIN_API_KEY=...
    JELLYFIN_USER_ID=...   (optional; auto-resolved from /Users if absent)

Usage:
    python scripts/export_jellyfin.py [--out-dir data/catalog]
"""

from __future__ import annotations

# 100ns ticks per minute (Jellyfin stores runtimes in .NET ticks)
TICKS_PER_MINUTE = 600_000_000
CAST_LIMIT = 10


def _people_of_type(item: dict, person_type: str) -> list[str]:
    """Names of People entries of the given type, in response (billing) order."""
    return [
        p["Name"]
        for p in item.get("People", [])
        if p.get("Type") == person_type and p.get("Name")
    ]


def normalize_item(item: dict) -> dict:
    """Map one raw Jellyfin /Items entry to the spec Section 4.1 movie schema."""
    provider_ids = item.get("ProviderIds", {}) or {}
    tmdb_raw = provider_ids.get("Tmdb")
    ticks = item.get("RunTimeTicks")
    directors = _people_of_type(item, "Director")
    user_data = item.get("UserData", {}) or {}
    return {
        "title": item.get("Name"),
        "year": item.get("ProductionYear"),
        "tmdb_id": int(tmdb_raw) if tmdb_raw else None,
        "imdb_id": provider_ids.get("Imdb"),
        "overview": item.get("Overview"),
        "genres": item.get("Genres", []) or [],
        "director": ", ".join(directors) if directors else None,
        "cast": _people_of_type(item, "Actor")[:CAST_LIMIT],
        "runtime_min": ticks // TICKS_PER_MINUTE if ticks else None,
        "tmdb_rating": item.get("CommunityRating"),
        "language": item.get("OriginalLanguage") or "en",
        "user_data": {
            "watched": bool(user_data.get("Played", False)),
            "play_count": int(user_data.get("PlayCount", 0)),
            "is_favorite": bool(user_data.get("IsFavorite", False)),
            "last_played_at": user_data.get("LastPlayedDate"),
            # Jellyfin exposes no numeric per-user rating; kept for schema stability
            "user_rating": None,
        },
    }
```

```python
import argparse
import json
import sys
import tempfile
from datetime import datetime, timezone
from pathlib import Path

import requests
from dotenv import dotenv_values

ENV_PATH = Path.home() / ".config" / "moviepicker" / ".env"
ITEM_FIELDS = "Overview,Genres,People,ProviderIds,CommunityRating,RunTimeTicks"


def _headers(api_key: str) -> dict:
    return {"X-Emby-Token": api_key}


def resolve_user_id(base_url: str, api_key: str) -> str:
    """Return the first Jellyfin user's id (single-user server assumption)."""
    resp = requests.Session().get(
        f"{base_url}/Users", headers=_headers(api_key), timeout=30
    )
    resp.raise_for_status()
    users = resp.json()
    if not users:
        raise RuntimeError("Jellyfin returned no users; cannot resolve user id")
    return users[0]["Id"]


def fetch_all_items(
    base_url: str, api_key: str, user_id: str, page_size: int = 100
) -> list[dict]:
    """Fetch every movie item, paginating by TotalRecordCount."""
    session = requests.Session()
    items: list[dict] = []
    start = 0
    while True:
        resp = session.get(
            f"{base_url}/Items",
            headers=_headers(api_key),
            params={
                "IncludeItemTypes": "Movie",
                "Recursive": "true",
                "Fields": ITEM_FIELDS,
                "userId": user_id,  # required for per-user UserData in the response
                "StartIndex": start,
                "Limit": page_size,
            },
            timeout=60,
        )
        resp.raise_for_status()
        payload = resp.json()
        items.extend(payload["Items"])
        start += page_size
        if start >= payload["TotalRecordCount"]:
            return items


def export_catalog(
    base_url: str, api_key: str, user_id: str | None, out_dir: Path
) -> Path:
    """Export the catalog to out_dir/jellyfin-YYYY-MM-DD.json atomically.

    Writes to a tempfile in the same directory and renames on success, so a
    failed export can never truncate or corrupt a previous good file, and
    latest.json only ever points at a complete export.
    """
    out_dir.mkdir(parents=True, exist_ok=True)
    if user_id is None:
        user_id = resolve_user_id(base_url, api_key)
    raw_items = fetch_all_items(base_url, api_key, user_id)
    envelope = {
        "source": "jellyfin",
        "exported_at": datetime.now(timezone.utc).isoformat(),
        "movies": [normalize_item(item) for item in raw_items],
    }
    date_stamp = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    final_path = out_dir / f"jellyfin-{date_stamp}.json"
    fd, tmp_name = tempfile.mkstemp(dir=out_dir, suffix=".tmp")
    tmp_path = Path(tmp_name)
    try:
        with open(fd, "w", encoding="utf-8") as f:
            json.dump(envelope, f, ensure_ascii=False, indent=2)
        tmp_path.rename(final_path)
    finally:
        tmp_path.unlink(missing_ok=True)
    latest = out_dir / "latest.json"
    latest.unlink(missing_ok=True)
    latest.symlink_to(final_path.name)
    return final_path


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--out-dir", type=Path, default=Path("data/catalog"))
    args = parser.parse_args()
    env = dotenv_values(ENV_PATH)
    base_url, api_key = env.get("JELLYFIN_URL"), env.get("JELLYFIN_API_KEY")
    if not base_url or not api_key:
        print(f"Missing JELLYFIN_URL / JELLYFIN_API_KEY in {ENV_PATH}", file=sys.stderr)
        return 1
    try:
        path = export_catalog(base_url, api_key, env.get("JELLYFIN_USER_ID"), args.out_dir)
    except (requests.RequestException, RuntimeError) as exc:
```

```python
"""Idempotent catalog-JSON to database ingest (spec Section 4, stage 2)."""

from __future__ import annotations

import json
import sys
from dataclasses import dataclass, field
from datetime import datetime
from pathlib import Path

from sqlalchemy.orm import Session

from .models import Movie, User, WatchHistory
from ..utils.logger import get_logger

logger = get_logger(__name__)


@dataclass
class IngestReport:
    """Counts and warnings from one ingest run."""

    added: int = 0
    updated: int = 0
    skipped: int = 0
    warnings: list[str] = field(default_factory=list)


def _parse_jellyfin_datetime(value: str | None) -> datetime | None:
    """Parse Jellyfin's ISO timestamps, which carry 7 fractional digits."""
    if not value:
        return None
    cleaned = value.rstrip("Z")
    if "." in cleaned:  # Python handles at most 6 fractional digits
        head, frac = cleaned.split(".", 1)
        cleaned = f"{head}.{frac[:6]}"
    try:
        return datetime.fromisoformat(cleaned)
    except ValueError:
        return None


def _get_or_create_user(session: Session, username: str) -> User:
    user = session.query(User).filter(User.username == username).first()
    if user is None:
        user = User(username=username, email=f"{username}@moviepicker.local")
        session.add(user)
        session.flush()
    return user


def _find_existing(session: Session, record: dict) -> Movie | None:
    if record.get("tmdb_id"):
        movie = session.query(Movie).filter(Movie.tmdb_id == record["tmdb_id"]).first()
        if movie:
            return movie
    return (
        session.query(Movie)
        .filter(Movie.title == record["title"], Movie.year == record.get("year"))
        .first()
    )


def _apply_fields(movie: Movie, record: dict) -> None:
    movie.title = record["title"]
    movie.year = record.get("year")
    movie.tmdb_id = record.get("tmdb_id")
    movie.overview = record.get("overview")
    movie.genres = ", ".join(record.get("genres") or []) or None
    movie.director = record.get("director")
    movie.cast = ", ".join(record.get("cast") or []) or None
    movie.rating = record.get("tmdb_rating")
    movie.runtime = record.get("runtime_min")
    movie.language = record.get("language")


def _upsert_watch_history(
    session: Session, user: User, movie: Movie, user_data: dict
) -> None:
    if not user_data.get("watched"):
        return
    row = (
        session.query(WatchHistory)
        .filter(
            WatchHistory.user_id == user.id, WatchHistory.movie_id == movie.id
        )
        .first()
    )
    if row is None:
        row = WatchHistory(user_id=user.id, movie_id=movie.id)
        session.add(row)
    row.is_favorite = bool(user_data.get("is_favorite", False))
    row.play_count = max(int(user_data.get("play_count", 1)), 1)
    watched_at = _parse_jellyfin_datetime(user_data.get("last_played_at"))
    if watched_at is not None:
        row.watched_at = watched_at


def ingest_catalog(
    session: Session, payload: dict, username: str = "default"
) -> IngestReport:
    """Upsert an export envelope into the database. Safe to re-run."""
    report = IngestReport()
    user = _get_or_create_user(session, username)
    for record in payload.get("movies", []):
        if not record.get("title"):
            report.skipped += 1
            report.warnings.append(f"missing title, skipped: {record!r:.120}")
            continue
        movie = _find_existing(session, record)
        if movie is None:
            movie = Movie()
            session.add(movie)
            report.added += 1
        else:
            report.updated += 1
        _apply_fields(movie, record)
        session.flush()  # movie.id needed for the watch-history upsert
        _upsert_watch_history(session, user, movie, record.get("user_data") or {})
    session.flush()
    logger.info(
        "Ingest complete: %d added, %d updated, %d skipped",
        report.added,
        report.updated,
        report.skipped,
    )
    return report


def main() -> int:
    if len(sys.argv) != 2:
        print("usage: python -m src.core.ingest <catalog.json>", file=sys.stderr)
        return 2
    from .database import db_manager  # deferred: import-time engine setup

    payload = json.loads(Path(sys.argv[1]).read_text(encoding="utf-8"))
    with db_manager.get_session() as session:
        report = ingest_catalog(session, payload)
    print(
        f"added={report.added} updated={report.updated} skipped={report.skipped}"
    )
    for warning in report.warnings:
        print(f"warning: {warning}", file=sys.stderr)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Build TF-IDF and semantic embedding artifacts (spec Section 4, stage 3).

Artifacts land in data/embeddings/ (gitignored):
    tfidf_matrix.npz, vectorizer.pkl        (TF-IDF)
    semantic_embeddings.npy                 (sentence-transformer)
    embedding_index.json                    (row -> movie_id, shared by both)
"""

from __future__ import annotations

import json
import pickle
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import scipy.sparse
from sklearn.feature_extraction.text import TfidfVectorizer
from sqlalchemy.orm import Session

from ..text import movie_to_document
from ...models import Movie
from ....utils.logger import get_logger

logger = get_logger(__name__)

DEFAULT_ARTIFACTS_DIR = Path("data/embeddings")


def _catalog_documents(session: Session) -> tuple[list[int], list[str]]:
    """All movies as (ids, documents), ordered by id so row order is stable."""
    movies = session.query(Movie).order_by(Movie.id).all()
    return [m.id for m in movies], [movie_to_document(m) for m in movies]


def _write_index(out_dir: Path, movie_ids: list[int]) -> None:
    index = {
        "movie_ids": movie_ids,
        "built_at": datetime.now(timezone.utc).isoformat(),
        "count": len(movie_ids),
    }
    (out_dir / "embedding_index.json").write_text(json.dumps(index))


def load_index(out_dir: Path) -> list[int]:
    """Row-order movie ids for the artifacts in out_dir."""
    return json.loads((out_dir / "embedding_index.json").read_text())["movie_ids"]


def build_tfidf(session: Session, out_dir: Path = DEFAULT_ARTIFACTS_DIR) -> None:
    """Fit a TF-IDF vectorizer over the catalog and persist matrix + vectorizer."""
    out_dir.mkdir(parents=True, exist_ok=True)
    movie_ids, docs = _catalog_documents(session)
    vectorizer = TfidfVectorizer(stop_words="english", max_features=20_000)
    matrix = vectorizer.fit_transform(docs)
    scipy.sparse.save_npz(out_dir / "tfidf_matrix.npz", matrix)
    with open(out_dir / "vectorizer.pkl", "wb") as f:
        pickle.dump(vectorizer, f)
    _write_index(out_dir, movie_ids)
    logger.info("Built TF-IDF artifacts for %d movies -> %s", len(movie_ids), out_dir)


def build_semantic(
    session: Session,
    out_dir: Path = DEFAULT_ARTIFACTS_DIR,
    model_name: str = "all-MiniLM-L6-v2",
) -> None:
    """Encode the catalog with a sentence-transformer and persist embeddings.

    sentence-transformers (and torch) are only needed here, at build time;
    serving loads the .npy directly. Hence the lazy import.
    """
    from sentence_transformers import SentenceTransformer

    out_dir.mkdir(parents=True, exist_ok=True)
    movie_ids, docs = _catalog_documents(session)
    model = SentenceTransformer(model_name)
    embeddings = model.encode(
        docs, normalize_embeddings=True, show_progress_bar=False
    )
    np.save(out_dir / "semantic_embeddings.npy", np.asarray(embeddings))
    _write_index(out_dir, movie_ids)
    logger.info(
        "Built semantic artifacts (%s) for %d movies -> %s",
        model_name,
        len(movie_ids),
        out_dir,
    )
```
