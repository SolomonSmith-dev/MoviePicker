"""The shared recommender seam: protocol, result type, errors (spec 3.1)."""

from __future__ import annotations

from dataclasses import dataclass
from typing import Protocol, runtime_checkable

from sqlalchemy.orm import Session


@dataclass
class Recommendation:
    """One ranked recommendation. score is min-max normalized to [0, 1]."""

    movie_id: int
    score: float
    reason: str


class RecommenderError(Exception):
    """Base class for recommender failures."""


class EmbeddingArtifactsMissingError(RecommenderError):
    """Vector artifacts are absent; run `moviepicker build-embeddings`."""


class NoPreferenceSignalError(RecommenderError):
    """The user has no watch history to build a preference profile from."""


@runtime_checkable
class RecommenderProtocol(Protocol):
    """Every strategy implements this; callers depend on nothing else."""

    name: str

    def fit(self, db: Session) -> None: ...

    def recommend(
        self,
        user_id: int,
        k: int = 10,
        exclude_watched: bool = True,
    ) -> list[Recommendation]: ...


def minmax_normalize(scores: list[float]) -> list[float]:
    """Map scores to [0, 1]; a constant list maps to all 1.0 (all equally best)."""
    if not scores:
        return []
    lo, hi = min(scores), max(scores)
    if hi == lo:
        return [1.0] * len(scores)
    return [(s - lo) / (hi - lo) for s in scores]
```

"""Implicit-feedback preference weights (Hu, Koren, Volinsky 2008 x recency).

weight = log(1 + play_count + favorite_bonus) * exp(-days_since_last_watched / 365)

At single-user scale the weighting choice is within eval noise; it is used
because it is what production implicit-feedback systems do (spec Section 5).
"""

from __future__ import annotations

import math
from datetime import datetime

from sqlalchemy.orm import Session

from ..models import WatchHistory
from .protocol import NoPreferenceSignalError

# A row with no timestamp gets one full decay unit rather than full freshness,
# so undated history can never dominate dated history.
DAYS_WHEN_UNDATED = 365.0
DECAY_DAYS = 365.0


def implicit_weight(
    play_count: int, is_favorite: bool, days_since_last_watched: float
) -> float:
    """Confidence-style weight for one watched movie."""
    favorite_bonus = 1.0 if is_favorite else 0.0
    confidence = math.log1p(max(play_count, 1) + favorite_bonus)
    recency = math.exp(-days_since_last_watched / DECAY_DAYS)
    return confidence * recency


def liked_movie_weights(
    session: Session, user_id: int, now: datetime | None = None
) -> dict[int, float]:
    """movie_id -> weight over every watched movie (binary 'liked' per spec 6)."""
    now = now or datetime.utcnow()
    rows = (
        session.query(WatchHistory).filter(WatchHistory.user_id == user_id).all()
    )
    if not rows:
        raise NoPreferenceSignalError(
            f"user {user_id} has no watch history; cannot build a profile"
        )
    weights: dict[int, float] = {}
    for row in rows:
        days = (
            (now - row.watched_at).total_seconds() / 86_400
            if row.watched_at
            else DAYS_WHEN_UNDATED
        )
        weights[row.movie_id] = implicit_weight(
            row.play_count or 1, bool(row.is_favorite), max(days, 0.0)
        )
    return weights


def watched_movie_ids(session: Session, user_id: int) -> set[int]:
    """Ids of every movie the user has a watch-history row for."""
    rows = (
        session.query(WatchHistory.movie_id)
        .filter(WatchHistory.user_id == user_id)
        .all()
    )
    return {movie_id for (movie_id,) in rows}
```

"""Baseline 0: the pre-v3 rule-based engine wrapped in the protocol.

src/core/recommendation.py is intentionally untouched; this adapter exists
so the old engine and the new vector recommenders are interchangeable.
"""

from __future__ import annotations

from sqlalchemy.orm import Session

from ..recommendation import RecommendationEngine, recommendation_engine
from .protocol import Recommendation


class RuleRecommender:
    """Adapter: RecommendationEngine -> RecommenderProtocol."""

    name = "rule"

    def __init__(self, engine: RecommendationEngine | None = None):
        # Injection point for the eval harness; production uses the global.
        self._engine = engine or recommendation_engine

    def fit(self, db: Session) -> None:
        """No-op: the wrapped engine queries the database on every call."""

    def recommend(
        self,
        user_id: int,
        k: int = 10,
        exclude_watched: bool = True,
    ) -> list[Recommendation]:
        movies = self._engine.get_recommendations(
            user_id=user_id, limit=k, exclude_watched=exclude_watched
        )
        count = len(movies)
        # The engine returns an ordering without scores; encode rank as score
        # so cross-strategy output is comparable (1.0 for the top slot).
        return [
            Recommendation(
                movie_id=movie.id,
                score=(count - i) / count,
                reason="rule-based match on your genre/director/actor preferences",
            )
            for i, movie in enumerate(movies)
        ]
```

"""Shared machinery for artifact-backed vector recommenders.

The user profile is a preference-weighted mean of the liked movies' catalog
vectors, so recommend-time needs no vectorizer or encoder: TF-IDF and
sentence-transformer strategies differ only in which matrix file they load.
"""

from __future__ import annotations

from pathlib import Path

import numpy as np
from sklearn.metrics.pairwise import cosine_similarity
from sqlalchemy.orm import Session

from .embeddings.build import DEFAULT_ARTIFACTS_DIR, load_index
from .preference import liked_movie_weights, watched_movie_ids
from .protocol import (
    EmbeddingArtifactsMissingError,
    Recommendation,
    minmax_normalize,
)


class VectorRecommenderBase:
    """Cosine top-k over prebuilt catalog vectors. Subclasses set the matrix."""

    name: str
    artifact_filename: str

    def __init__(self, artifacts_dir: Path | None = None):
        # Resolved in the body, not as a default argument, so tests (and any
        # future config) can swap DEFAULT_ARTIFACTS_DIR at call time.
        self._artifacts_dir = Path(artifacts_dir) if artifacts_dir else DEFAULT_ARTIFACTS_DIR
        self._session: Session | None = None
        self._matrix = None
        self._movie_ids: list[int] = []
        self._row_of: dict[int, int] = {}

    def _load_matrix(self, path: Path):
        raise NotImplementedError

    def fit(self, db: Session) -> None:
        """Load artifacts and keep the session for recommend-time queries."""
        matrix_path = self._artifacts_dir / self.artifact_filename
        index_path = self._artifacts_dir / "embedding_index.json"
        if not matrix_path.exists() or not index_path.exists():
            raise EmbeddingArtifactsMissingError(
                f"missing {matrix_path.name} in {self._artifacts_dir}; "
                "run `moviepicker build-embeddings` first"
            )
        self._matrix = self._load_matrix(matrix_path)
        self._movie_ids = load_index(self._artifacts_dir)
        self._row_of = {mid: row for row, mid in enumerate(self._movie_ids)}
        self._session = db

    def recommend(
        self,
        user_id: int,
        k: int = 10,
        exclude_watched: bool = True,
    ) -> list[Recommendation]:
        if self._session is None or self._matrix is None:
            raise EmbeddingArtifactsMissingError(
                f"{self.name} recommender used before fit()"
            )
        weights = liked_movie_weights(self._session, user_id)
        # Weighted mean of liked rows = the user profile vector. Movies
        # ingested after the last artifact build have no row; skip them.
        rows, row_weights = [], []
        for movie_id, weight in weights.items():
            row = self._row_of.get(movie_id)
            if row is not None:
                rows.append(row)
                row_weights.append(weight)
        if not rows:
            raise EmbeddingArtifactsMissingError(
                "no liked movie is present in the artifacts; rebuild with "
                "`moviepicker build-embeddings`"
            )
        w = np.asarray(row_weights, dtype=np.float64)
        w /= w.sum()
        profile = np.asarray(
            sum(wi * self._matrix[ri] for wi, ri in zip(w, rows))
        ).reshape(1, -1)
        sims = cosine_similarity(profile, self._matrix).ravel()

        excluded = watched_movie_ids(self._session, user_id) if exclude_watched else set()
        ranked = [
            (self._movie_ids[row], float(sims[row]))
            for row in np.argsort(-sims)
            if self._movie_ids[row] not in excluded
        ][:k]
        scores = minmax_normalize([s for _, s in ranked])
        return [
            Recommendation(
                movie_id=movie_id,
                score=score,
                reason=f"cosine similarity {raw:.2f} to your liked-movie "
                f"profile ({self.name})",
            )
            for (movie_id, raw), score in zip(ranked, scores)
        ]
```

"""Baseline 1: TF-IDF content recommender over prebuilt artifacts."""

from __future__ import annotations

from pathlib import Path

import scipy.sparse

"""Recommender implementations behind a shared protocol (v3.0 AI/ML pivot)."""

from __future__ import annotations

from pathlib import Path

from .embed import EmbedRecommender
from .protocol import Recommendation, RecommenderProtocol
from .rule import RuleRecommender
from .tfidf import TfidfRecommender

STRATEGIES: tuple[str, ...] = ("rule", "tfidf", "embed")

_CLASSES = {
    "rule": RuleRecommender,
    "tfidf": TfidfRecommender,
    "embed": EmbedRecommender,
}

__all__ = [
    "STRATEGIES",
    "Recommendation",
    "RecommenderProtocol",
    "get_recommender",
]


def get_recommender(name: str, artifacts_dir: Path | None = None) -> RecommenderProtocol:
    """Instantiate a strategy by name. artifacts_dir only applies to vector ones."""
    if name not in _CLASSES:
        raise ValueError(f"unknown strategy {name!r}; expected one of {STRATEGIES}")
    if name == "rule":
        return _CLASSES[name]()
    if artifacts_dir is not None:
        return _CLASSES[name](artifacts_dir=artifacts_dir)
    return _CLASSES[name]()

def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="moviepicker",
        description="MoviePicker CLI. Run with no arguments for the interactive menu.",
    )
    sub = parser.add_subparsers(dest="command")

    recommend = sub.add_parser("recommend", help="print ranked recommendations")
    recommend.add_argument(
        "--strategy", choices=[*STRATEGIES, "all"], default="embed"
    )
    recommend.add_argument("--k", type=int, default=10)
    recommend.add_argument("--user-id", dest="user_id", type=int, default=1)

    ingest = sub.add_parser("ingest", help="ingest an exported catalog JSON")
    ingest.add_argument("path", help="path to jellyfin-YYYY-MM-DD.json or latest.json")

    build = sub.add_parser("build-embeddings", help="build vector artifacts")
    build.add_argument(
        "--skip-semantic",
        action="store_true",
        help="only build TF-IDF (no sentence-transformers required)",
    )
    return parser


def _print_strategy_table(
    session: Session, strategy: str, k: int, user_id: int
) -> None:
    recommender = get_recommender(strategy)
    recommender.fit(session)
    results = recommender.recommend(user_id=user_id, k=k)
    titles = {
        m.id: m for m in session.query(Movie).filter(
            Movie.id.in_([r.movie_id for r in results])
        )
    }
    table = Table(title=f"strategy: {strategy}")
    table.add_column("#", justify="right")
    table.add_column("title")
    table.add_column("score", justify="right")
    table.add_column("reason")
    for i, rec in enumerate(results, start=1):
        movie = titles.get(rec.movie_id)
        label = f"{movie.title} ({movie.year})" if movie else str(rec.movie_id)
        table.add_row(str(i), label, f"{rec.score:.2f}", rec.reason)
    console.print(table)


def run_recommend(
    strategy: str, k: int, user_id: int, session: Session | None = None
) -> int:
    """Print top-k for one strategy, or all three side by side."""
    strategies = list(STRATEGIES) if strategy == "all" else [strategy]

    def _run(active: Session) -> int:
        for name in strategies:
            try:
                _print_strategy_table(active, name, k, user_id)
            except RecommenderError as exc:
                console.print(f"[red]{name}: {exc}[/red]")
                if strategy != "all":
                    return 1
        return 0

    if session is not None:
        return _run(session)
    with db_manager.get_session() as active:
        return _run(active)


def run_ingest(path: str) -> int:
    payload = json.loads(Path(path).read_text(encoding="utf-8"))
    with db_manager.get_session() as session:
        report = ingest_catalog(session, payload)
    console.print(
        f"added={report.added} updated={report.updated} skipped={report.skipped}"
    )
    for warning in report.warnings:
        console.print(f"[yellow]warning:[/yellow] {warning}")
    return 0


def run_build_embeddings(skip_semantic: bool = False) -> int:
    with db_manager.get_session() as session:
        build_tfidf(session)
        if skip_semantic:
            console.print("Skipped semantic build (--skip-semantic)")
        else:
            try:
                build_semantic(session)
            except ImportError:
                console.print(
                    "[red]sentence-transformers not installed; "
                    "run `pip install -r requirements-ml.txt` "
                    "or pass --skip-semantic[/red]"
                )
                return 1
    console.print("Artifacts written to data/embeddings/")
    return 0
